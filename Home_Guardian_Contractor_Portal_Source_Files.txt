import {db} from '../../../../lib/server';
export async function GET(req:Request){
 if(!process.env.CRON_SECRET||req.headers.get('authorization')!==`Bearer ${process.env.CRON_SECRET}`)return new Response('Unauthorized',{status:401});
 if(!process.env.RESEND_API_KEY||!process.env.NOTIFICATION_FROM_EMAIL)return Response.json({error:'Email provider not configured'},{status:503});
 const client=db();
 const {data:batch,error}=await client.rpc('lease_notification_batch',{p_limit:40});
 if(error)return Response.json({error:'Could not lease notification batch'},{status:500});
 let sent=0,failed=0;
 for(const job of batch||[]){
  try{
   // Resolve email from trusted Auth database, not contractor-controlled profile text.
   const {data:user,error:lookupError}=await client.auth.admin.getUserById(job.recipient_id);
   if(lookupError||!user?.user?.email)throw new Error('Recipient email unavailable');
   const res=await fetch('https://api.resend.com/emails',{method:'POST',headers:{
    Authorization:`Bearer ${process.env.RESEND_API_KEY}`,'Content-Type':'application/json',
    'Idempotency-Key':job.event_key
   },body:JSON.stringify({from:process.env.NOTIFICATION_FROM_EMAIL,to:[user.user.email],subject:job.subject,text:job.body})});
   if(!res.ok)throw new Error(`Provider returned ${res.status}`);
   await client.from('notification_outbox').update({status:'sent',delivered_at:new Date().toISOString(),lease_until:null,last_error:null}).eq('id',job.id).eq('status','processing').eq('attempts',job.attempts);
   sent++;
  }catch(e){
   const attempts=Number(job.attempts);
   const backoffMinutes=Math.min(360,Math.pow(2,attempts));
   await client.from('notification_outbox').update({
    status:attempts>=8?'failed':'pending',lease_until:null,
    next_attempt_at:new Date(Date.now()+backoffMinutes*60000).toISOString(),
    last_error:String(e).slice(0,200)
   }).eq('id',job.id).eq('status','processing').eq('attempts',job.attempts);
   failed++;
  }
 }
 return Response.json({leased:(batch||[]).length,sent,failed});
}

import Stripe from 'stripe';
import {db,stripe} from '../../../../lib/server';
export async function POST(req:Request){
 const sig=req.headers.get('stripe-signature');if(!sig)return new Response('Missing signature',{status:400});
 let event:Stripe.Event;
 try{event=stripe().webhooks.constructEvent(await req.text(),sig,process.env.STRIPE_WEBHOOK_SECRET!)}catch{return new Response('Invalid signature',{status:400})}
 if(event.type==='checkout.session.completed'){
   const s=event.data.object as Stripe.Checkout.Session;
   if(s.mode==='subscription'&&s.subscription&&s.client_reference_id){
     const sub=await stripe().subscriptions.retrieve(s.subscription as string);
     await sync(sub,s.client_reference_id);
   }
 }
 if(event.type==='customer.subscription.updated'||event.type==='customer.subscription.deleted'){
   const sub=event.data.object as Stripe.Subscription;
   const userId=sub.metadata.user_id;
   if(userId)await sync(sub,userId);
 }
 return Response.json({received:true});
}
async function sync(sub:Stripe.Subscription,userId:string){
 const item=sub.items.data[0];const paidThrough=item?.current_period_end?new Date(item.current_period_end*1000).toISOString():null;
 const plan=sub.metadata.plan==='annual'?'annual':'monthly';
 const {error}=await db().from('memberships').upsert({user_id:userId,stripe_subscription_id:sub.id,status:sub.status,plan,paid_through:paidThrough,updated_at:new Date().toISOString()},{onConflict:'user_id'});
 if(error)throw error;
}

# Home Guardian starter

A Next.js + Supabase + Stripe foundation for paid homeowner membership, guided maintenance onboarding, and scheduled reminder records.

## Setup
1. `npm install`
2. Copy `.env.example` to `.env.local` and fill values.
3. Run `supabase/schema.sql` in the Supabase SQL editor.
4. Configure Stripe webhook endpoint `/api/stripe/webhook` for `checkout.session.completed`, `customer.subscription.updated`, and `customer.subscription.deleted`.
5. Run `npm run dev`.
6. Schedule `GET /api/cron/reminders` daily using a scheduler, with `Authorization: Bearer <CRON_SECRET>`.

## Scope
- Authenticated Supabase sessions via bearer access token in API requests.
- Checkout and subscription status synced by Stripe webhook.
- Guided setup builds recommended maintenance reminders.
- Due reminders recorded in notification inbox; email/push delivery is a future integration.
- Repair request submission requires an active paid membership and is protected in the database.

This starter requires your own Supabase/Stripe credentials and deployment to process real payments. Do not expose service-role or Stripe secret keys to the browser.

## Exclusive contractor assignments (1,500-member readiness)
1. Run `supabase/exclusive_contractors.sql` after the base schema.
2. Admin provisions contractor rows with `approved=true`, `trade`, and `service_area` (server-side only).
3. Homeowner request POST now requires `service_area` as well as trade and description. Update the frontend form to send the homeowner's verified service area.
4. Contractors call `GET /api/contractor/opportunities` with a Supabase bearer token, then `POST /api/contractor/claim` with `{ "requestId": 123 }`.
5. Only the homeowner and assigned contractor can access `GET/POST /api/requests/123/messages`.
6. The database's assignment primary key and atomic claim RPC prevent simultaneous double claims. Admin reassignment requires a separate audited server workflow (not included).

### Capacity notes
Designed for 1,500 registered members, **not load-tested or production certified**. Use managed Postgres connection pooling, indexes, pagination, queue-based reminders, Stripe webhook idempotency, rate limits, monitoring, and concurrency/load testing before launch. The existing daily reminders route should be refactored to batch work for larger populations. Add authenticated admin approval, more granular service-area matching, abuse protection, and homeowner notification delivery before production.

## Management dashboard add-on
Visit `/admin` after setting up Supabase and applying `schema.sql`, `exclusive_contractors.sql`, and `secure_reassignment.sql` in that order.
Add administrator user IDs to `home_guardian_admins` using a privileged SQL editor.
The dashboard supports aggregate membership metrics, paginated contractor approvals, paginated work orders, and reopen-and-notify using the existing locked database function.
**Development authentication:** `/admin` currently accepts a Supabase user access token entered manually; it is NOT a production-ready login flow. Only server-side admin-verified endpoints access the service-role database client. Replace this with secure session authentication before launch.
Contractor alerts are **in-app records only**, not delivered email/push. No deployment, load test, or security audit was executed. See `tests/admin-security-checklist.md`.

## Secure operations extension
Apply `supabase/operations_security.sql` after the earlier migrations.
- Assignment history: `GET /api/admin/history`
- Audit history: `GET /api/admin/audit`
- Message reads: transactional, assignment-checked RPC; no direct two-query TOCTOU authorization
- Approvals and reopens: transactional audit events
- Job reopened: outbox events are created atomically; `/api/cron/deliver` delivers via Resend with retries, leased workers and idempotency keys

-- Locking request rows makes claims, releases, and message inserts mutually exclusive.
create or replace function public.claim_repair(p_request_id bigint,p_contractor_id uuid)
returns boolean language plpgsql security definer set search_path=public as $$
declare r public.repair_requests%rowtype;
begin
 select * into r from public.repair_requests where id=p_request_id for update;
 if not found or r.status <> 'open' then return false; end if;
 if not exists(select 1 from public.contractors c where c.user_id=p_contractor_id and c.approved and c.trade=r.trade and c.service_area=r.service_area) then return false; end if;
 insert into public.repair_assignments(request_id,contractor_id) values(p_request_id,p_contractor_id) on conflict do nothing;
 if not found then return false; end if;
 update public.repair_requests set status='assigned' where id=p_request_id;
 return true;
end $$;
-- Contractor can decline their own job; an administrator may reopen any assignment.
-- The original contractor is excluded from this reopening notification batch.
create or replace function public.reopen_repair(p_request_id bigint,p_actor_id uuid)
returns integer language plpgsql security definer set search_path=public as $$
declare r public.repair_requests%rowtype; old_contractor uuid; n integer;
begin
 select * into r from public.repair_requests where id=p_request_id for update;
 if not found then raise exception 'Request not found'; end if;
 select contractor_id into old_contractor from public.repair_assignments where request_id=p_request_id;
 if old_contractor is null or r.status <> 'assigned' then raise exception 'Request not assigned'; end if;
 if p_actor_id <> old_contractor and not exists(select 1 from public.home_guardian_admins where user_id=p_actor_id) then raise exception 'Not authorized'; end if;
 -- Delete previous conversation so the next contractor cannot read old messages.
 delete from public.repair_messages where request_id=p_request_id;
 delete from public.repair_assignments where request_id=p_request_id;
 update public.repair_requests set status='open' where id=p_request_id;
 insert into public.contractor_alerts(contractor_id,request_id,event_type)
 select c.user_id,p_request_id,'reopened' from public.contractors c
 where c.approved and c.trade=r.trade and c.service_area=r.service_area and c.user_id<>old_contractor;
 get diagnostics n=row_count;
 return n;
end $$;
-- Message authorization and insert must happen inside the same row lock as reassignment.
create or replace function public.send_repair_message(p_request_id bigint,p_sender_id uuid,p_body text)
returns bigint language plpgsql security definer set search_path=public as $$
declare r public.repair_requests%rowtype; assigned uuid; message_id bigint;
begin
 if length(trim(coalesce(p_body,''))) not between 1 and 2000 then raise exception 'Invalid message'; end if;
 select * into r from public.repair_requests where id=p_request_id for update;
 if not found or r.status <> 'assigned' then raise exception 'Request unavailable'; end if;
 select contractor_id into assigned from public.repair_assignments where request_id=p_request_id;
 if assigned is null or (p_sender_id <> assigned and p_sender_id <> r.user_id) then raise exception 'Access denied'; end if;
 insert into public.repair_messages(request_id,sender_id,body) values(p_request_id,p_sender_id,trim(p_body)) returning id into message_id;
 return message_id;
end $$;

end $$;
-- Lock request row for all assignment transitions; history and alerts commit atomically.
create or replace function public.claim_repair(p_request_id bigint,p_contractor_id uuid)
returns boolean language plpgsql security definer set search_path=public as $$
declare r public.repair_requests%rowtype;
begin
 select * into r from public.repair_requests where id=p_request_id for update;
 if not found or r.status<>'open' then return false; end if;
 if not exists(select 1 from public.contractors c where c.user_id=p_contractor_id and c.approved and c.trade=r.trade and c.service_area=r.service_area) then return false; end if;
 insert into public.repair_assignments(request_id,contractor_id) values(p_request_id,p_contractor_id) on conflict do nothing;
 if not found then return false; end if;
 update public.repair_requests set status='assigned' where id=p_request_id;
 insert into public.contractor_assignment_history(request_id,contractor_id,action,actor_id)
 values(p_request_id,p_contractor_id,'accepted',p_contractor_id);
 return true;
end $$;
create or replace function public.reopen_repair(p_request_id bigint,p_actor_id uuid)
returns integer language plpgsql security definer set search_path=public as $$
declare r public.repair_requests%rowtype; old_contractor uuid; n integer; is_admin boolean;
begin
 select * into r from public.repair_requests where id=p_request_id for update;
 if not found or r.status<>'assigned' then raise exception 'Request unavailable'; end if;
 select contractor_id into old_contractor from public.repair_assignments where request_id=p_request_id;
 if old_contractor is null then raise exception 'Assignment missing'; end if;
 select exists(select 1 from public.home_guardian_admins where user_id=p_actor_id) into is_admin;
 if p_actor_id<>old_contractor and not is_admin then raise exception 'Not authorized'; end if;
 insert into public.contractor_assignment_history(request_id,contractor_id,action,actor_id)
 values(p_request_id,old_contractor,case when is_admin and p_actor_id<>old_contractor then 'admin_reopened' else 'declined' end,p_actor_id);
 if is_admin and p_actor_id<>old_contractor then
  insert into public.admin_audit_logs(actor_id,action,target_type,target_id,details)
  values(p_actor_id,'reopen_request','repair_request',p_request_id::text,jsonb_build_object('previous_contractor',old_contractor));
 end if;
 delete from public.repair_messages where request_id=p_request_id;
 delete from public.repair_assignments where request_id=p_request_id;
 update public.repair_requests set status='open' where id=p_request_id;
 -- No homeowner private details in these alerts; only the work order reference.
 insert into public.contractor_alerts(contractor_id,request_id,event_type)
 select c.user_id,p_request_id,'reopened' from public.contractors c
 where c.approved and c.trade=r.trade and c.service_area=r.service_area and c.user_id<>old_contractor;
 get diagnostics n=row_count;
 insert into public.notification_outbox(recipient_id,request_id,kind,event_key,subject,body)
 select c.user_id,p_request_id,'job_reopened',
 'job-reopened:'||p_request_id||':'||extract(epoch from clock_timestamp())::bigint||':'||c.user_id,
 'Home Guardian: job available','An eligible work order is available in your contractor dashboard.'
 from public.contractors c where c.approved and c.trade=r.trade and c.service_area=r.service_area and c.user_id<>old_contractor;
 return n;
end $$;
create or replace function public.set_contractor_approval(p_actor_id uuid,p_contractor_id uuid,p_approved boolean)
returns boolean language plpgsql security definer set search_path=public as $$
begin
 if not exists(select 1 from public.home_guardian_admins where user_id=p_actor_id) then raise exception 'Not authorized'; end if;
 update public.contractors set approved=p_approved where user_id=p_contractor_id;
 if not found then return false; end if;
 insert into public.admin_audit_logs(actor_id,action,target_type,target_id,details)
 values(p_actor_id,case when p_approved then 'approve_contractor' else 'suspend_contractor' end,
 'contractor',p_contractor_id::text,jsonb_build_object('approved',p_approved));
 return true;
end $$;
-- Queue claiming via SKIP LOCKED to allow parallel workers.
create or replace function public.lease_notification_batch(p_limit int default 50)
returns setof public.notification_outbox language plpgsql security definer set search_path=public as $$
begin
 return query with selected as (
  select o.id from public.notification_outbox o
  where ((o.status='pending' and o.next_attempt_at<=now())
  or (o.status='processing' and o.lease_until<now()))
  and o.attempts<8
  order by o.created_at for update skip locked limit least(greatest(p_limit,1),100)
 ), updated as (
  update public.notification_outbox o set status='processing',attempts=o.attempts+1,lease_until=now()+interval '2 minutes'
  from selected s where o.id=s.id returning o.*
 ) select * from updated;
end $$;
